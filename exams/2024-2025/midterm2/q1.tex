\section*{Question 1}
A long, straight, solid cylindrical conductor of radius $R$ carries a steady total current $I$ in the $+\hat z$ direction.  The volume current density varies with radial distance $s$ from the axis as
\begin{equation*}
    \vec{J}(s) = 
    \begin{cases}
    J_0\,\dfrac{s}{R}\hat{z}, & 0 \le s \le R,\\[6pt]
    0, & s > R.
    \end{cases}
\end{equation*}
    \begin{enumerate}
      \item[(a)]  Determine \(J_0\) in terms of the total current \(I\) and the radius \(R\). --- \textit{0.4 pts}
      \item[(b)]  Using Ampère’s law, find the magnetic field \(\vec{B}(s)\) (magnitude and direction) for
      \begin{enumerate}
        \item[(i)] \(s < R\), --- \textit{0.2 pts}
        \item[(ii)] \(s > R\). --- \textit{0.2 pts}
      \end{enumerate}
      \item[(c)]  Sketch \(B(s)\) versus \(s\) for \(0 \le s \le 2R\). --- \textit{0.2 pts}
    \end{enumerate}
