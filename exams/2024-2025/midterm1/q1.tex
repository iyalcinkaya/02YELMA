\section*{Question 1}
A solid insulating sphere of radius $R$ carries a spherically symmetric, non-uniform charge distribution given by
\begin{equation*}
    \rho(r) = \rho_0 \left(1 - \frac{r}{R}\right), \quad 0 < r < R,
\end{equation*}
and $\rho(r)=0$ for $r \geq R$. Here, $\rho_0$ is a constant. 
\begin{enumerate}
    \item [(a)] Calculate the electric field $\vec{E}(r)$ as a function of the distance $r$ from the center of the sphere, for $0 < r < R$ and $r \geq R$. --- \textit{0.9 pts}
    \item [(b)] Do the expresions for $\vec{E}(r)$ you derived for inside and outside the sphere agree at $r=R$? --- \textit{0.1 pts}
\end{enumerate}
